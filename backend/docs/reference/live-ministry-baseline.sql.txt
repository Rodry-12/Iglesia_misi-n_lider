CREATE FUNCTION iglesia.validar_asignacion_proceso() RETURNS trigger
    LANGUAGE plpgsql
    SET search_path TO ''
    AS $$
DECLARE
    v_participante_valido BOOLEAN;
    v_responsable_valido BOOLEAN;
    v_proceso_activo BOOLEAN;
BEGIN

    -- Impedir que una persona sea su propio responsable.
    IF NEW.persona_id = NEW.responsable_persona_id THEN
        RAISE EXCEPTION
            'El participante y el responsable deben ser personas diferentes';
    END IF;

    -- Verificar participante activo en la sede.
    SELECT EXISTS (
        SELECT 1
        FROM iglesia.persona p
        INNER JOIN iglesia.persona_sede ps
            ON ps.persona_id = p.id
        WHERE p.id = NEW.persona_id
          AND p.activa = TRUE
          AND ps.sede_id = NEW.sede_id
          AND ps.fecha_fin IS NULL
    )
    INTO v_participante_valido;

    IF NOT v_participante_valido THEN
        RAISE EXCEPTION
            'El participante no pertenece activamente a la sede';
    END IF;

    -- Verificar responsable activo en la misma sede.
    SELECT EXISTS (
        SELECT 1
        FROM iglesia.persona p
        INNER JOIN iglesia.persona_sede ps
            ON ps.persona_id = p.id
        WHERE p.id = NEW.responsable_persona_id
          AND p.activa = TRUE
          AND ps.sede_id = NEW.sede_id
          AND ps.fecha_fin IS NULL
    )
    INTO v_responsable_valido;

    IF NOT v_responsable_valido THEN
        RAISE EXCEPTION
            'El responsable no pertenece activamente a la sede';
    END IF;

    -- Comprobar que el proceso y su categoría estén activos.
    SELECT EXISTS (
        SELECT 1
        FROM iglesia.proceso pr
        INNER JOIN iglesia.categoria_proceso cp
            ON cp.id = pr.categoria_id
        WHERE pr.id = NEW.proceso_id
          AND pr.activo = TRUE
          AND cp.activa = TRUE
    )
    INTO v_proceso_activo;

    IF NOT v_proceso_activo THEN
        RAISE EXCEPTION
            'El proceso o su categoría no están activos';
    END IF;

    RETURN NEW;
END;
$$;

CREATE FUNCTION iglesia.validar_estado_fechas() RETURNS trigger
    LANGUAGE plpgsql
    SET search_path TO ''
    AS $$
DECLARE
    v_codigo VARCHAR(30);
BEGIN
    SELECT codigo
    INTO v_codigo
    FROM iglesia.estado_proceso
    WHERE id = NEW.estado_id;

    IF v_codigo IS NULL THEN
        RAISE EXCEPTION 'El estado del proceso no existe';
    END IF;

    -- Estado pendiente
    IF v_codigo = 'PENDING' THEN
        IF NEW.fecha_inicio IS NOT NULL
           OR NEW.fecha_finalizacion IS NOT NULL THEN
            RAISE EXCEPTION
                'Un proceso pendiente no puede tener fechas de inicio o finalización';
        END IF;

    -- Estado en proceso
    ELSIF v_codigo = 'IN_PROGRESS' THEN
        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := NULL;

    -- Estado suspendido
    ELSIF v_codigo = 'SUSPENDED' THEN
        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := NULL;

    -- Estado completado
    ELSIF v_codigo = 'COMPLETED' THEN
        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := COALESCE(
            NEW.fecha_finalizacion,
            NOW()
        );

    -- Estado retirado
    ELSIF v_codigo = 'WITHDRAWN' THEN
        NEW.fecha_finalizacion := COALESCE(
            NEW.fecha_finalizacion,
            NOW()
        );

    ELSE
        RAISE EXCEPTION
            'Código de estado no permitido: %',
            v_codigo;
    END IF;

    IF NEW.fecha_inicio IS NOT NULL
       AND NEW.fecha_inicio < NEW.fecha_asignacion THEN
        RAISE EXCEPTION
            'La fecha de inicio no puede ser anterior a la asignación';
    END IF;

    IF NEW.fecha_finalizacion IS NOT NULL
       AND NEW.fecha_inicio IS NOT NULL
       AND NEW.fecha_finalizacion < NEW.fecha_inicio THEN
        RAISE EXCEPTION
            'La fecha de finalización no puede ser anterior al inicio';
    END IF;

    RETURN NEW;
END;
$$;

CREATE FUNCTION iglesia.validar_proceso_duplicado() RETURNS trigger
    LANGUAGE plpgsql
    SET search_path TO ''
    AS $$
DECLARE
    v_estado_final BOOLEAN;
BEGIN
    SELECT ep.es_final
    INTO v_estado_final
    FROM iglesia.estado_proceso ep
    WHERE ep.id = NEW.estado_id;

    IF v_estado_final IS NULL THEN
        RAISE EXCEPTION 'Estado de proceso inválido';
    END IF;

    -- Serializa operaciones concurrentes para esta persona y proceso.
    PERFORM pg_advisory_xact_lock(
        hashtextextended(
            NEW.persona_id::text || ':' || NEW.proceso_id::text,
            0
        )
    );

    IF NOT v_estado_final AND EXISTS (
        SELECT 1
        FROM iglesia.persona_proceso pp
        INNER JOIN iglesia.estado_proceso ep
            ON ep.id = pp.estado_id
        WHERE pp.persona_id = NEW.persona_id
          AND pp.proceso_id = NEW.proceso_id
          AND ep.es_final = FALSE
          AND pp.id <> NEW.id
    ) THEN
        RAISE EXCEPTION
            'La persona ya tiene una asignación abierta de este proceso';
    END IF;

    RETURN NEW;
END;
$$;
