-- ============================================================
-- 02. SEDE
-- ============================================================

CREATE TABLE iglesia.sede (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    codigo VARCHAR(20) NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    direccion VARCHAR(250),
    telefono VARCHAR(30),

    activa BOOLEAN NOT NULL DEFAULT TRUE,

    fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    fecha_actualizacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT uq_sede_codigo
        UNIQUE (codigo),

    CONSTRAINT ck_sede_codigo
        CHECK (LENGTH(TRIM(codigo)) > 0),

    CONSTRAINT ck_sede_nombre
        CHECK (LENGTH(TRIM(nombre)) > 0)
);


-- ============================================================
-- 03. PERSONA
-- ============================================================

CREATE TABLE iglesia.persona (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    nombres VARCHAR(150) NOT NULL,
    apellidos VARCHAR(150) NOT NULL,

    telefono VARCHAR(30),
    correo VARCHAR(255),
    fecha_nacimiento DATE,

    activa BOOLEAN NOT NULL DEFAULT TRUE,

    fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    fecha_actualizacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT ck_persona_nombres
        CHECK (LENGTH(TRIM(nombres)) > 0),

    CONSTRAINT ck_persona_apellidos
        CHECK (LENGTH(TRIM(apellidos)) > 0)
);


-- ============================================================
-- 04. PERSONA_DOCUMENTO
-- ============================================================

CREATE TABLE iglesia.persona_documento (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    persona_id UUID NOT NULL
        REFERENCES iglesia.persona(id)
        ON DELETE RESTRICT,

    tipo_documento VARCHAR(30) NOT NULL,
    numero_documento VARCHAR(50) NOT NULL,
    pais_emisor CHAR(2) NOT NULL DEFAULT 'CO',

    vigente BOOLEAN NOT NULL DEFAULT TRUE,

    fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    fecha_finalizacion TIMESTAMPTZ,

    CONSTRAINT uq_persona_documento
        UNIQUE (
            pais_emisor,
            tipo_documento,
            numero_documento
        ),

    CONSTRAINT ck_documento_numero
        CHECK (LENGTH(TRIM(numero_documento)) > 0),

    CONSTRAINT ck_documento_tipo
        CHECK (LENGTH(TRIM(tipo_documento)) > 0),

    CONSTRAINT ck_documento_vigencia
        CHECK (
            (
                vigente = TRUE
                AND fecha_finalizacion IS NULL
            )
            OR
            (
                vigente = FALSE
                AND fecha_finalizacion IS NOT NULL
            )
        )
);

CREATE UNIQUE INDEX ux_persona_documento_vigente
ON iglesia.persona_documento(persona_id)
WHERE vigente = TRUE;

CREATE INDEX ix_persona_documento_persona
ON iglesia.persona_documento(persona_id);


-- ============================================================
-- 05. PERSONA_SEDE
-- ============================================================

CREATE TABLE iglesia.persona_sede (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    persona_id UUID NOT NULL
        REFERENCES iglesia.persona(id)
        ON DELETE RESTRICT,

    sede_id UUID NOT NULL
        REFERENCES iglesia.sede(id)
        ON DELETE RESTRICT,

    fecha_inicio TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    fecha_fin TIMESTAMPTZ,

    motivo_salida TEXT,

    fecha_creacion TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CONSTRAINT ck_persona_sede_fechas
        CHECK (
            fecha_fin IS NULL
            OR fecha_fin >= fecha_inicio
        )
);

CREATE UNIQUE INDEX ux_persona_sede_activa
ON iglesia.persona_sede(persona_id)
WHERE fecha_fin IS NULL;

CREATE INDEX ix_persona_sede_sede
ON iglesia.persona_sede(sede_id);

-- ============================================================
-- 13. FUNCION: VALIDAR ASIGNACION
-- ============================================================

CREATE OR REPLACE FUNCTION iglesia.validar_asignacion_proceso()
RETURNS TRIGGER
LANGUAGE plpgsql
SET search_path = ''
AS $$
DECLARE
    v_participante_valido BOOLEAN;
    v_responsable_valido BOOLEAN;
    v_proceso_activo BOOLEAN;
BEGIN

    IF NEW.persona_id = NEW.responsable_persona_id THEN
        RAISE EXCEPTION
            'El participante y el responsable deben ser personas diferentes';
    END IF;

    SELECT EXISTS (
        SELECT 1
        FROM iglesia.persona p
        INNER JOIN iglesia.persona_sede ps
            ON ps.persona_id = p.id
        WHERE p.id = NEW.persona_id
          AND p.activa = TRUE
          AND ps.sede_id = NEW.sede_id
          AND ps.fecha_fin IS NULL
    )
    INTO v_participante_valido;

    IF NOT v_participante_valido THEN
        RAISE EXCEPTION
            'El participante no pertenece activamente a la sede';
    END IF;

    SELECT EXISTS (
        SELECT 1
        FROM iglesia.persona p
        INNER JOIN iglesia.persona_sede ps
            ON ps.persona_id = p.id
        WHERE p.id = NEW.responsable_persona_id
          AND p.activa = TRUE
          AND ps.sede_id = NEW.sede_id
          AND ps.fecha_fin IS NULL
    )
    INTO v_responsable_valido;

    IF NOT v_responsable_valido THEN
        RAISE EXCEPTION
            'El responsable no pertenece activamente a la sede';
    END IF;

    SELECT EXISTS (
        SELECT 1
        FROM iglesia.proceso pr
        INNER JOIN iglesia.categoria_proceso cp
            ON cp.id = pr.categoria_id
        WHERE pr.id = NEW.proceso_id
          AND pr.activo = TRUE
          AND cp.activa = TRUE
    )
    INTO v_proceso_activo;

    IF NOT v_proceso_activo THEN
        RAISE EXCEPTION
            'El proceso o su categoria no estan activos';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_validar_asignacion_proceso
BEFORE INSERT OR UPDATE OF
    persona_id,
    responsable_persona_id,
    sede_id,
    proceso_id
ON iglesia.persona_proceso
FOR EACH ROW
EXECUTE FUNCTION iglesia.validar_asignacion_proceso();


-- ============================================================
-- 14. FUNCION: EVITAR PROCESOS ABIERTOS DUPLICADOS
-- ============================================================

CREATE OR REPLACE FUNCTION iglesia.validar_proceso_duplicado()
RETURNS TRIGGER
LANGUAGE plpgsql
SET search_path = ''
AS $$
DECLARE
    v_estado_final BOOLEAN;
BEGIN

    SELECT ep.es_final
    INTO v_estado_final
    FROM iglesia.estado_proceso ep
    WHERE ep.id = NEW.estado_id;

    IF v_estado_final IS NULL THEN
        RAISE EXCEPTION 'Estado de proceso invalido';
    END IF;

    PERFORM pg_catalog.pg_advisory_xact_lock(
        pg_catalog.hashtextextended(
            NEW.persona_id::text || ':' || NEW.proceso_id::text,
            0
        )
    );

    IF NOT v_estado_final AND EXISTS (
        SELECT 1
        FROM iglesia.persona_proceso pp
        INNER JOIN iglesia.estado_proceso ep
            ON ep.id = pp.estado_id
        WHERE pp.persona_id = NEW.persona_id
          AND pp.proceso_id = NEW.proceso_id
          AND ep.es_final = FALSE
          AND pp.id <> NEW.id
    ) THEN
        RAISE EXCEPTION
            'La persona ya tiene una asignacion abierta de este proceso';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_validar_proceso_duplicado
BEFORE INSERT OR UPDATE OF
    persona_id,
    proceso_id,
    estado_id
ON iglesia.persona_proceso
FOR EACH ROW
EXECUTE FUNCTION iglesia.validar_proceso_duplicado();


-- ============================================================
-- 15. FUNCION: VALIDAR ESTADOS Y FECHAS
-- ============================================================

CREATE OR REPLACE FUNCTION iglesia.validar_estado_fechas()
RETURNS TRIGGER
LANGUAGE plpgsql
SET search_path = ''
AS $$
DECLARE
    v_codigo VARCHAR(30);
BEGIN

    SELECT codigo
    INTO v_codigo
    FROM iglesia.estado_proceso
    WHERE id = NEW.estado_id;

    IF v_codigo IS NULL THEN
        RAISE EXCEPTION
            'El estado del proceso no existe';
    END IF;

    IF v_codigo = 'PENDING' THEN

        IF NEW.fecha_inicio IS NOT NULL
           OR NEW.fecha_finalizacion IS NOT NULL THEN
            RAISE EXCEPTION
                'Un proceso pendiente no puede tener fechas de inicio o finalizacion';
        END IF;

    ELSIF v_codigo = 'IN_PROGRESS' THEN

        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := NULL;

    ELSIF v_codigo = 'SUSPENDED' THEN

        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := NULL;

    ELSIF v_codigo = 'COMPLETED' THEN

        NEW.fecha_inicio := COALESCE(
            NEW.fecha_inicio,
            NOW()
        );

        NEW.fecha_finalizacion := COALESCE(
            NEW.fecha_finalizacion,
            NOW()
        );

    ELSIF v_codigo = 'WITHDRAWN' THEN

        NEW.fecha_finalizacion := COALESCE(
            NEW.fecha_finalizacion,
            NOW()
        );

    ELSE

        RAISE EXCEPTION
            'Codigo de estado no permitido: %',
            v_codigo;
    END IF;

    IF NEW.fecha_inicio IS NOT NULL
       AND NEW.fecha_inicio < NEW.fecha_asignacion THEN
        RAISE EXCEPTION
            'La fecha de inicio no puede ser anterior a la asignacion';
    END IF;

    IF NEW.fecha_finalizacion IS NOT NULL
       AND NEW.fecha_inicio IS NOT NULL
       AND NEW.fecha_finalizacion < NEW.fecha_inicio THEN
        RAISE EXCEPTION
            'La fecha de finalizacion no puede ser anterior al inicio';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER trg_validar_estado_fechas
BEFORE INSERT OR UPDATE OF estado_id
ON iglesia.persona_proceso
FOR EACH ROW
EXECUTE FUNCTION iglesia.validar_estado_fechas();

-- 26. CONSULTAS DE REFERENCIA PARA EL BACKEND

-- CONSULTAR PERSONAS POR SEDE

SELECT
    p.id,
    p.nombres,
    p.apellidos,
    p.telefono,
    p.correo,
    s.nombre AS sede
FROM iglesia.persona p
INNER JOIN iglesia.persona_sede ps
    ON ps.persona_id = p.id
INNER JOIN iglesia.sede s
    ON s.id = ps.sede_id
WHERE ps.fecha_fin IS NULL
ORDER BY p.nombres, p.apellidos;


-- CONSULTAR PROCESOS ASIGNADOS

SELECT
    pp.id,
    p.nombres || ' ' || p.apellidos AS participante,
    pr.nombre AS proceso,
    r.nombres || ' ' || r.apellidos AS responsable,
    s.nombre AS sede,
    ep.nombre AS estado,
    pp.fecha_asignacion,
    pp.fecha_inicio,
    pp.fecha_finalizacion
FROM iglesia.persona_proceso pp
INNER JOIN iglesia.persona p
    ON p.id = pp.persona_id
INNER JOIN iglesia.persona r
    ON r.id = pp.responsable_persona_id
INNER JOIN iglesia.proceso pr
    ON pr.id = pp.proceso_id
INNER JOIN iglesia.sede s
    ON s.id = pp.sede_id
INNER JOIN iglesia.estado_proceso ep
    ON ep.id = pp.estado_id
ORDER BY p.nombres, pr.nombre;


-- CONSULTAR CANTIDAD DE PROCESOS POR ESTADO

SELECT
    ep.codigo,
    ep.nombre,
    COUNT(pp.id) AS cantidad
FROM iglesia.estado_proceso ep
LEFT JOIN iglesia.persona_proceso pp
    ON pp.estado_id = ep.id
GROUP BY ep.id, ep.codigo, ep.nombre
ORDER BY ep.codigo;


-- CONSULTAR PERMISOS POR ROL

SELECT
    r.codigo AS rol,
    p.modulo,
    p.codigo AS permiso,
    p.nombre
FROM iglesia.rol_permiso rp
INNER JOIN iglesia.rol r
    ON r.id = rp.rol_id
INNER JOIN iglesia.permiso p
    ON p.id = rp.permiso_id
ORDER BY r.codigo, p.modulo, p.codigo;
